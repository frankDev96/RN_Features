
To create boilerplate folders & files
-------------------------------

mkdir -p src/{components,screens,navigation,utils,assets/{images,icons}} &&
cd src &&
touch components/CustomButton.js screens/{HomeScreen.js,LoginScreen.js} navigation/AppNavigator.js utils/helpers.js App.js index.js &&
cd ../ &&
touch package.json
